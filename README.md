## AUTOMATED STATS WORKFLOW

Create this file in the same repository:

`.github/workflows/readme-stats.yml`

```yaml
name: Update GitHub Stats

on:
  schedule:
    - cron: "17 3 * * *"
  workflow_dispatch:
  push:
    branches:
      - main

permissions:
  contents: write

jobs:
  update:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout profile
        uses: actions/checkout@v4

      - name: Create output directory
        run: mkdir -p profile

      - name: Generate stats
        uses: soulteary/github-readme-stats-action@v1.0.0
        with:
          card: stats
          options: 'username=${{ github.repository_owner }}&show_icons=true&hide_border=true&include_all_commits=true&hide_rank=true&theme=transparent'
          path: profile/stats.svg
          token: ${{ secrets.GITHUB_TOKEN }}

      - name: Generate languages
        uses: soulteary/github-readme-stats-action@v1.0.0
        with:
          card: top-langs
          options: 'username=${{ github.repository_owner }}&layout=compact&langs_count=8&hide_border=true&theme=transparent'
          path: profile/top-langs.svg
          token: ${{ secrets.GITHUB_TOKEN }}

      - name: Commit generated cards
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add profile/stats.svg profile/top-langs.svg
          git commit -m "chore: Update profile stats" || exit 0
          git push
```
